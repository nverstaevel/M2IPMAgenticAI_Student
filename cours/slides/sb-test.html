Tester l’API sans navigateur

@SpringBootTest
class ApiTest {
    // MockMvc construit avec Spring Security (@BeforeEach)
    @Test
    @WithMockUser(roles = "LECTURE")
    void rechercheParNom() throws Exception {
        mvc.perform(get("/api/proprietaires").param("nom", "brume"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$[0].nom").value("Brumevent"));
    }
}
